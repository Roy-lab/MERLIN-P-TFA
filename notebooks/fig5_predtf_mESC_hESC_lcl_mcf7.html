# Rename naming formats
cell_types = 'mESC'.split()
def clean_GS_src(GS_src):
    data = GS_src.split("_")
    if data[0] in cell_types:
        return '_'.join(data[1:])
    else:
        return '_'.join(data)

# Strip cell type labels from gold standard names
scores_mesc_rnaseq.GS_src = scores_mesc_rnaseq.GS_src.apply(clean_GS_src)

# and renanme the three primary gold standards
scores_mesc_rnaseq.tfa = scores_mesc_rnaseq.tfa.replace({'noTFA': 'Exp', 
                                       'Unreg_TFA': 'TFA 0.000',
                                       'Reg_TFA': 'Best TFA'})

TFA_order  = ['MERLIN_Exp','MERLIN_TFA 0.000','MERLIN_Best TFA',
              'MERLIN-P_Exp','MERLIN-P_TFA 0.000','MERLIN-P_Best TFA']

scores_mesc_rnaseq['bar_sort'] = scores_mesc_rnaseq[['algo', 'tfa']].apply(lambda x: '_'.join(x), axis=1)

scores_mesc_rnaseq = scores_mesc_rnaseq[[  "dataset","algo", "parameter", 
                                         "tfa","GS_src","n_edges", "pred_TF", "bar_sort"]]

scores_mesc_rnaseq.sort_values(by="bar_sort", key=lambda column: column.map(lambda e: TFA_order.index(e)), inplace=True)

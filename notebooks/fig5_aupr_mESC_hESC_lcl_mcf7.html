# Rename naming formats
cell_types = 'mESC'.split()
def clean_GS_src(GS_src):
    data = GS_src.split("_")
    if data[0] in cell_types:
        return '_'.join(data[1:])
    else:
        return '_'.join(data)

# Strip cell type labels from gold standard names
scores_mesc_rnaseq.GS_src = scores_mesc_rnaseq.GS_src.apply(clean_GS_src)

# and renanme the three primary gold standards
scores_mesc_rnaseq.tfa = scores_mesc_rnaseq.tfa.replace({'noTFA': 'Exp',
